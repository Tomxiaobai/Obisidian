	func getVideoInstanceUpdateModelWhenStatusInvalid(ctx context.Context, instance *model.PromoteInstance) (*model.PromoteInstanceUpdateModel, werror.Error) {  
    itemInfo, err := item.Ins().GetItemInfo(ctx, instance.ContentID)  
    if err != nil {  
       return nil, err  
    }  
    if itemInfo == nil {  
       return nil, nil  
    }  
    if itemInfo.Status == item.StatusUserDel {  
       reqStatus := model.PromoteInstanceStatusOffline  
       return &model.PromoteInstanceUpdateModel{  
          PromoteID: instance.ID,  
          ContentID: instance.ContentID,  
          Status:    &reqStatus,  
          EndTime:   time.Unix(itemInfo.ModifyTime, 0),  
       }, nil  
    }  
    if itemInfo.Status != item.StatusPublic {  
       log.V2.Warn().With(ctx).Str("getVideoInstanceUpdateModelWhenStatusInvalid status invalid").KV("instance", jsonx.ToString(instance)).Emit()  
       return nil, nil  
    }  
    return nil, nil  
}
